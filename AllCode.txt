// User.java - common base for Admin, Vendor and Customer
// Abstraction: you cannot create a plain User, only one of the three roles.
// Encapsulation: the password is private and can only be checked, never read.
public abstract class User {
    private String name;
    private String loginId;
    private String password;

    public User(String name, String loginId, String password) {
        this.name = name;
        this.loginId = loginId;
        this.password = password;
    }

    public String getName() { return name; }
    public String getLoginId() { return loginId; }

    public boolean checkPassword(String input) {
        return password.equals(input);
    }

    // Polymorphism: each role gives its own answer
    public abstract String getRole();
}

// Book.java - common base for all books
// Abstraction: 'abstract' means you cannot create a plain Book,
// only a Fiction, Crime, Horror or Comedy book.
// Encapsulation: fields are private, accessed only through methods.
public abstract class Book {
    private int id;
    private String title;
    private String author;
    private double price;
    private int quantity;
    private Vendor vendor;

    public Book(int id, String title, String author, double price, int quantity, Vendor vendor) {
        this.id = id;
        this.title = title;
        this.author = author;
        this.price = price;
        this.quantity = quantity;
        this.vendor = vendor;
    }

    public int getId() { return id; }
    public String getTitle() { return title; }
    public String getAuthor() { return author; }
    public double getPrice() { return price; }
    public int getQuantity() { return quantity; }
    public Vendor getVendor() { return vendor; }

    public void reduceStock(int qty) { quantity = quantity - qty; }
    public void addStock(int qty) { quantity = quantity + qty; }

    // Polymorphism: every category answers these differently
    public abstract String getCategory();
    public abstract double getDiscountPercent();
    public abstract String getNote();

    // Common code shared by all categories
    public double getFinalPrice() {
        return price - (price * getDiscountPercent() / 100);
    }
}

import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.ArrayList;

// Store.java - connects the Store, its Users, Books and Orders. All the logic is here.
public class Store {
    private String name;
    private Admin admin = new Admin("Administrator", "admin", "admin123");
    private ArrayList<Vendor> vendors = new ArrayList<>();
    private ArrayList<Customer> customers = new ArrayList<>();
    private ArrayList<Book> books = new ArrayList<>();
    private ArrayList<Order> orders = new ArrayList<>();
    private ArrayList<String> log = new ArrayList<>();
    private int nextBookId = 1;
    private int nextOrderId = 1;

    public Store(String name) {
        this.name = name;
    }

    public String getName() { return name; }
    public ArrayList<Vendor> getVendors() { return vendors; }
    public ArrayList<Customer> getCustomers() { return customers; }
    public ArrayList<Order> getOrders() { return orders; }
    public ArrayList<String> getLog() { return log; }

    // ---------- Login and registration ----------

    public User login(String loginId, String password) throws StoreException {
        if (loginId == null || loginId.trim().isEmpty() || password == null || password.isEmpty()) {
            throw new StoreException("Enter your login ID and password.");
        }
        loginId = loginId.trim();
        // Polymorphism: Admin, Vendor and Customer are all treated as User here
        ArrayList<User> all = new ArrayList<>();
        all.add(admin);
        all.addAll(vendors);
        all.addAll(customers);
        for (User u : all) {
            if (u.getLoginId().equalsIgnoreCase(loginId) && u.checkPassword(password)) {
                return u;
            }
        }
        throw new StoreException("Invalid login ID or password.");
    }

    public void registerCustomer(String name, String email, String password) throws StoreException {
        if (name == null || name.trim().isEmpty() || email == null || email.trim().isEmpty()
                || password == null) {
            throw new StoreException("Name, email and password are required.");
        }
        email = email.trim().toLowerCase();
        if (!email.matches("[^@\\s]+@[^@\\s]+\\.[^@\\s]+")) {
            throw new StoreException("Enter a valid email address.");
        }
        if (password.length() < 4) {
            throw new StoreException("Password must have at least 4 characters.");
        }
        if (loginIdTaken(email)) {
            throw new StoreException("This email is already registered.");
        }
        customers.add(new Customer(name.trim(), email, password));
    }

    // Admin creates vendors
    public void addVendor(String name, String loginId, String password,
                          String phone, String city) throws StoreException {
        if (name == null || name.trim().isEmpty() || loginId == null || loginId.trim().isEmpty()
                || city == null || city.trim().isEmpty() || password == null) {
            throw new StoreException("Vendor name, login ID, password and city are required.");
        }
        if (password.length() < 4) {
            throw new StoreException("Password must have at least 4 characters.");
        }
        if (phone == null || !phone.trim().matches("\\d{7,15}")) {
            throw new StoreException("Phone must contain only digits (7 to 15).");
        }
        if (loginIdTaken(loginId.trim())) {
            throw new StoreException("This login ID is already used.");
        }
        vendors.add(new Vendor(name.trim(), loginId.trim(), password, phone.trim(), city.trim()));
    }

    // ---------- Books ----------

    public void addBook(Vendor vendor, String category, String title, String author,
                        double price, int quantity) throws StoreException {
        if (title == null || title.trim().isEmpty() || author == null || author.trim().isEmpty()) {
            throw new StoreException("Title and author cannot be empty.");
        }
        if (price <= 0) {
            throw new StoreException("Price must be greater than 0.");
        }
        if (quantity < 0) {
            throw new StoreException("Quantity cannot be negative.");
        }
        if (category == null) {
            throw new StoreException("Please choose a category.");
        }

        title = title.trim();
        author = author.trim();
        Book b;
        switch (category) {
            case "Fiction": b = new Fiction(nextBookId, title, author, price, quantity, vendor); break;
            case "Crime":   b = new Crime(nextBookId, title, author, price, quantity, vendor); break;
            case "Horror":  b = new Horror(nextBookId, title, author, price, quantity, vendor); break;
            case "Comedy":  b = new Comedy(nextBookId, title, author, price, quantity, vendor); break;
            default: throw new StoreException("Unknown category: " + category);
        }
        books.add(b);
        nextBookId++;
        addLog("Vendor " + vendor.getName() + " added book '" + title + "' (" + category
                + ") with stock " + quantity + ".");
    }

    // A vendor can add stock only to their own books
    public void addStock(Vendor vendor, int bookId, int qty) throws StoreException {
        Book b = findBook(bookId);
        if (b == null || b.getVendor() != vendor) {
            throw new StoreException("You can only add stock to your own books.");
        }
        if (qty <= 0) {
            throw new StoreException("Stock quantity must be at least 1.");
        }
        b.addStock(qty);
        addLog("Vendor " + vendor.getName() + " added " + qty + " stock to '" + b.getTitle()
                + "'. Stock is now " + b.getQuantity() + ".");
    }

    public void deleteBook(int id) throws StoreException {
        Book b = findBook(id);
        if (b == null) {
            throw new StoreException("Book not found.");
        }
        books.remove(b);
        addLog("Admin deleted book '" + b.getTitle() + "' (vendor " + b.getVendor().getName() + ").");
    }

    // Empty keyword / empty category means "match all"
    public ArrayList<Book> search(String keyword, String category) {
        ArrayList<Book> result = new ArrayList<>();
        String k = keyword.toLowerCase();
        for (Book b : books) {
            boolean textMatch = b.getTitle().toLowerCase().contains(k)
                    || b.getAuthor().toLowerCase().contains(k);
            boolean categoryMatch = category.isEmpty() || b.getCategory().equals(category);
            if (textMatch && categoryMatch) {
                result.add(b);
            }
        }
        return result;
    }

    public ArrayList<Book> booksOf(Vendor vendor) {
        ArrayList<Book> result = new ArrayList<>();
        for (Book b : books) {
            if (b.getVendor() == vendor) {
                result.add(b);
            }
        }
        return result;
    }

    // ---------- Orders ----------

    public String placeOrder(Customer customer, int bookId, int qty) throws StoreException {
        Book b = findBook(bookId);
        if (b == null) {
            throw new StoreException("Book not available.");
        }
        if (qty <= 0) {
            throw new StoreException("Order quantity must be at least 1.");
        }
        if (qty > b.getQuantity()) {
            throw new StoreException("Only " + b.getQuantity() + " copies of '" + b.getTitle() + "' in stock.");
        }
        b.reduceStock(qty);
        double total = b.getFinalPrice() * qty;
        orders.add(new Order(nextOrderId, customer, b, qty, total));
        nextOrderId++;
        return "Order placed: " + qty + " x " + b.getTitle() + ". Total: Rs." + String.format("%.2f", total);
    }

    public ArrayList<Order> getOrdersOf(Customer customer) {
        ArrayList<Order> result = new ArrayList<>();
        for (Order o : orders) {
            if (o.getCustomer() == customer) {
                result.add(o);
            }
        }
        return result;
    }

    // ---------- Private helpers ----------

    private Book findBook(int id) {
        for (Book b : books) {
            if (b.getId() == id) {
                return b;
            }
        }
        return null;
    }

    private boolean loginIdTaken(String loginId) {
        if (admin.getLoginId().equalsIgnoreCase(loginId)) {
            return true;
        }
        for (Vendor v : vendors) {
            if (v.getLoginId().equalsIgnoreCase(loginId)) {
                return true;
            }
        }
        for (Customer c : customers) {
            if (c.getLoginId().equalsIgnoreCase(loginId)) {
                return true;
            }
        }
        return false;
    }

    // Newest entry goes first
    private void addLog(String text) {
        String time = LocalDateTime.now().format(DateTimeFormatter.ofPattern("dd MMM yyyy HH:mm"));
        log.add(0, time + "  -  " + text);
    }
}

    // Shows a different page depending on who is logged in
    static void home(HttpExchange ex) throws IOException {
        User user = currentUser(ex);
        Map<String, String> q = parse(ex.getRequestURI().getRawQuery());
        String msg = q.getOrDefault("msg", "");

        StringBuilder h = new StringBuilder();
        startPage(h, msg, user);

        if (user == null) {
            loginPage(h);
        } else if (user instanceof Admin) {
            adminPage(h);
        } else if (user instanceof Vendor) {
            vendorPage(h, (Vendor) user);
        } else {
            customerPage(h, (Customer) user, q.getOrDefault("q", ""), q.getOrDefault("cat", ""));
        }

        h.append("</body></html>");
        byte[] data = h.toString().getBytes(StandardCharsets.UTF_8);
        ex.getResponseHeaders().set("Content-Type", "text/html; charset=UTF-8");
        ex.sendResponseHeaders(200, data.length);
        try (OutputStream os = ex.getResponseBody()) {
            os.write(data);
        }
    }

            cell(h, b.getAuthor());
            cell(h, b.getCategory());
            cell(h, money(b.getPrice()));
            cell(h, (int) b.getDiscountPercent() + "%");
            cell(h, money(b.getFinalPrice()));
            cell(h, b.getQuantity());
            cell(h, b.getNote());
            if (b.getQuantity() > 0) {
                h.append("<td><form class='inline' method='post' action='/order'>");
                h.append("<input type='hidden' name='id' value='").append(b.getId()).append("'>");
                h.append("<input type='number' name='qty' value='1' min='1' max='")
                        .append(b.getQuantity()).append("' style='width:60px'>");
                h.append("<button>Order</button></form></td>");
            } else {
                h.append("<td>Out of stock</td>");
            }
            h.append("</tr>");
        }
        h.append("</table>");

        h.append("<h3>My Orders</h3><table>");
        headRow(h, "Order", "Book", "Quantity", "Total");
        for (Order o : store.getOrdersOf(c)) {
            h.append("<tr>");
            cell(h, o.getId());
            cell(h, o.getBook().getTitle());
            cell(h, o.getQuantity());
            cell(h, money(o.getTotal()));
            h.append("</tr>");
        }
        h.append("</table>");
    }

    static void login(HttpExchange ex) throws IOException {
        if (!isPost(ex)) { redirect(ex, ""); return; }
        Map<String, String> f = parse(body(ex));
        try {
            User u = store.login(f.get("loginId"), f.get("password"));
            String sid = UUID.randomUUID().toString();
            sessions.put(sid, u);
            ex.getResponseHeaders().add("Set-Cookie", "sid=" + sid + "; Path=/; HttpOnly");
            redirect(ex, "Welcome, " + u.getName() + " (" + u.getRole() + ").");
        } catch (StoreException e) {
            redirect(ex, e.getMessage());
        }
    }

    static void register(HttpExchange ex) throws IOException {
        if (!isPost(ex)) { redirect(ex, ""); return; }
        Map<String, String> f = parse(body(ex));
        try {
            store.registerCustomer(f.get("name"), f.get("email"), f.get("password"));
            redirect(ex, "Registration successful. Please log in.");
        } catch (StoreException e) {
            redirect(ex, e.getMessage());
        }
    }

    static void logout(HttpExchange ex) throws IOException {
        String sid = getSid(ex);
        if (sid != null) {
            sessions.remove(sid);
        }
        ex.getResponseHeaders().add("Set-Cookie", "sid=; Path=/; Max-Age=0");
        redirect(ex, "Logged out.");
    }

    static void addVendor(HttpExchange ex) throws IOException {
        User u = currentUser(ex);
        if (!isPost(ex) || !(u instanceof Admin)) { redirect(ex, "Please log in as Admin."); return; }
        Map<String, String> f = parse(body(ex));
        try {
            store.addVendor(f.get("name"), f.get("loginId"), f.get("password"), f.get("phone"), f.get("city"));
            redirect(ex, "Vendor added.");
        } catch (StoreException e) {
            redirect(ex, e.getMessage());
        }
    }

    static void deleteBook(HttpExchange ex) throws IOException {
        User u = currentUser(ex);
        if (!isPost(ex) || !(u instanceof Admin)) { redirect(ex, "Please log in as Admin."); return; }
        Map<String, String> f = parse(body(ex));
        try {
            store.deleteBook(Integer.parseInt(f.get("id")));
            redirect(ex, "Book deleted.");
        } catch (StoreException e) {
            redirect(ex, e.getMessage());
        } catch (NumberFormatException | NullPointerException e) {
            redirect(ex, "Incorrect input. Invalid book ID.");
        }
    }

    static void addBook(HttpExchange ex) throws IOException {
        User u = currentUser(ex);
        if (!isPost(ex) || !(u instanceof Vendor)) { redirect(ex, "Please log in as Vendor."); return; }
        Map<String, String> f = parse(body(ex));
        try {
            double price = Double.parseDouble(f.get("price"));
            int qty = Integer.parseInt(f.get("quantity"));
            store.addBook((Vendor) u, f.get("category"), f.get("title"), f.get("author"), price, qty);
            redirect(ex, "Book added.");
        } catch (StoreException e) {
            redirect(ex, e.getMessage());
        } catch (NumberFormatException | NullPointerException e) {
            redirect(ex, "Incorrect input. Please check price and quantity.");
        }
    }

    static void addStock(HttpExchange ex) throws IOException {
        User u = currentUser(ex);
        if (!isPost(ex) || !(u instanceof Vendor)) { redirect(ex, "Please log in as Vendor."); return; }
        Map<String, String> f = parse(body(ex));
        try {
            store.addStock((Vendor) u, Integer.parseInt(f.get("id")), Integer.parseInt(f.get("qty")));
            redirect(ex, "Stock added.");
        } catch (StoreException e) {
            redirect(ex, e.getMessage());
        } catch (NumberFormatException | NullPointerException e) {
            redirect(ex, "Incorrect input. Please enter a valid quantity.");
        }
    }

    static void order(HttpExchange ex) throws IOException {
        User u = currentUser(ex);
        if (!isPost(ex) || !(u instanceof Customer)) { redirect(ex, "Please log in as Customer."); return; }
        Map<String, String> f = parse(body(ex));
        try {
            int id = Integer.parseInt(f.get("id"));
            int qty = Integer.parseInt(f.get("qty"));
            redirect(ex, store.placeOrder((Customer) u, id, qty));
        } catch (StoreException e) {
            redirect(ex, e.getMessage());
        } catch (NumberFormatException | NullPointerException e) {
            redirect(ex, "Incorrect input. Please enter a valid quantity.");
        }
    }

    // =====================================================
    //  SMALL HELPERS
    // =====================================================

    static boolean isPost(HttpExchange ex) {
        return ex.getRequestMethod().equals("POST");
    }

    // Finds the session id from the browser cookie
    static String getSid(HttpExchange ex) {
        String cookie = ex.getRequestHeaders().getFirst("Cookie");
        if (cookie == null) {
            return null;
        }
        for (String part : cookie.split(";")) {
            String[] kv = part.trim().split("=", 2);
            if (kv.length == 2 && kv[0].equals("sid")) {
                return kv[1];
            }
        }
        return null;
    }

    // Returns the logged-in user, or null if nobody is logged in
    static User currentUser(HttpExchange ex) {
        String sid = getSid(ex);
        if (sid == null) {
            return null;
        }
        return sessions.get(sid);
    }

    static String body(HttpExchange ex) throws IOException {
        return new String(ex.getRequestBody().readAllBytes(), StandardCharsets.UTF_8);
    }

    // Turns "a=1&b=2" into a Map
    static Map<String, String> parse(String s) throws IOException {
        Map<String, String> map = new HashMap<>();
        if (s == null || s.isEmpty()) {
            return map;
        }
        for (String pair : s.split("&")) {
            String[] kv = pair.split("=", 2);
            String key = URLDecoder.decode(kv[0], "UTF-8");
            String value = kv.length > 1 ? URLDecoder.decode(kv[1], "UTF-8") : "";
            map.put(key, value);
        }
        return map;
    }

    // Sends the browser back to the home page with a message
    static void redirect(HttpExchange ex, String msg) throws IOException {
        ex.getResponseHeaders().set("Location", "/?msg=" + URLEncoder.encode(msg, "UTF-8"));
        ex.sendResponseHeaders(303, -1);
        ex.close();
    }

    static void headRow(StringBuilder h, String... names) {
        h.append("<tr>");
        for (String n : names) {
            h.append("<th>").append(n).append("</th>");
        }
        h.append("</tr>");
    }

    static void cell(StringBuilder h, Object value) {
        h.append("<td>").append(esc(String.valueOf(value))).append("</td>");
    }

    static String money(double d) {
        return "Rs." + String.format("%.2f", d);
    }

    // Stops typed text from breaking the HTML
    static String esc(String s) {
        return s.replace("&", "&amp;").replace("<", "&lt;")
                .replace(">", "&gt;").replace("'", "&#39;").replace("\"", "&quot;");
    }
